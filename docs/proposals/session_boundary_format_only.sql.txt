-- PROPOSAL ONLY. Never apply remotely before exact review/approval.
-- Reject sessionless/malformed JWT session IDs without adding privileged table access.
create or replace function public.sadeeman_session_active() returns boolean
language sql stable security invoker set search_path='' as $$
  select (select auth.uid()) is not null and case
    when coalesce(nullif(current_setting('request.jwt.claims',true),'')::jsonb->>'session_id','')
      ~ '^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$'
    then not exists(
      select 1 from public.sadeeman_session_revocations
      where session_id=(nullif(current_setting('request.jwt.claims',true),'')::jsonb->>'session_id')::uuid
    )
    else false
  end
$$;
-- Preserve guest/self read behavior; only the INSERT check becomes narrower.
alter policy own_sessions on public.sadeeman_session_revocations to authenticated
with check (
  owner_id=(select auth.uid()) and tenant_id=(select auth.uid())
  and session_id=(select case
    when coalesce(nullif(current_setting('request.jwt.claims',true),'')::jsonb->>'session_id','')
      ~ '^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$'
    then (nullif(current_setting('request.jwt.claims',true),'')::jsonb->>'session_id')::uuid
    else null end)
);
-- Trigger functions need no public direct-call surface. Existing triggers remain installed.
revoke execute on function sadeeman_private.guard_consent() from public,anon,authenticated;
revoke execute on function sadeeman_private.guard_memory() from public,anon,authenticated;
